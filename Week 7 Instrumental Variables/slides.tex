\begin{frame}{A Change of Strategy}
Week 6 assumed we had measured every confounder and worked on \emph{how} to
adjust. That assumption is usually false, and no diagnostic can rescue it.
\vskip3pt
\begin{alertblock}{The instrumental variables idea}
Give up on controlling for confounders. Instead, find a source of variation in
the treatment that is \emph{as good as randomly assigned}, and use only that
variation.
\end{alertblock}
\vskip3pt
\begin{itemize}\itemsep2pt
  \item We throw away most of the variation in $D$ and keep a sliver we can
        defend. Precision falls; credibility rises.
  \item The price is steep and often hidden: we no longer estimate the ATE, and
        the assumptions are as untestable as ignorability was, just different,
        and usually easier to argue about in substantive terms.
\end{itemize}
\end{frame}

\section[IV]{Instrumental Variable}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{What Goes Wrong, in Scalar Form\core}
We want $\beta$ in $Y_i = \alpha + \beta D_i + \epsilon_i$, but $D_i$ is
\alert{endogenous}, $\Cov(D_i,\epsilon_i)\neq 0$. OLS gives
\vskip-6pt
\begin{align*}
  \hat\beta = \frac{\Cov_n(D_i, Y_i)}{\Var_n(D_i)}
     &= \frac{\Cov_n(D_i,\ \alpha + \beta D_i + \epsilon_i)}{\Var_n(D_i)}
     && \text{\footnotesize substitute the model for $Y_i$} \\
     &= \beta + \frac{\Cov_n(D_i,\epsilon_i)}{\Var_n(D_i)}
     && \text{\footnotesize covariance is linear; the constant drops} \\
     &\pto \beta + \frac{\Cov(D_i,\epsilon_i)}{\Var(D_i)}
     && \text{\footnotesize LLN: sample moments to population moments}
\end{align*}
\vskip-5pt
The second term does not shrink with $n$: OLS is biased \emph{and} inconsistent.
\vskip0pt
\textbf{Three sources of $\Cov(D_i,\epsilon_i) \neq 0$}: \textbf{omitted variables}
(ability, family resources sit in $\epsilon$ and move with $D$; Week 1's OVB
formula); \textbf{simultaneity} ($Y$ also causes $D$: police and crime);
\textbf{selection} into the sample on a collider (Week 5).
\vskip1pt
\textbf{A different problem, measurement error in $D$}: classical error in $D$
shrinks the OLS slope toward zero, by the reliability ratio $\Var(D^*)/\Var(D)$.
A known direction, unlike the three above.
\end{frame}

\begin{frame}[allowframebreaks,t]{An Instrument\core}
An instrument $Z_i$ moves $D_i$ but is otherwise unrelated to $Y_i$.
\vskip-2pt
\begin{center}
\begin{tikzpicture}[scale=0.92,>=Stealth,
  every node/.style={circle,draw=dukeblue,fill=white,inner sep=1pt,minimum size=8mm,font=\scriptsize}]
  \node (z) at (0,0) {$Z$};  \node (d) at (2.4,0) {$D$};  \node (y) at (4.8,0) {$Y$};
  \node[draw=gray,dashed] (u) at (3.6,1.4) {$U$};
  \draw[->,dukeblue,thick] (z)--(d) node[draw=none,fill=none,midway,above,font=\scriptsize]{relevance};
  \draw[->,dukeblue,thick] (d)--(y);
  \draw[->,gray,dashed,thick] (u)--(d); \draw[->,gray,dashed,thick] (u)--(y);
  \draw[->,accent,dashed,thick] (z) to[bend right=28] node[draw=none,fill=none,midway,below,font=\scriptsize,text=accent]{must not exist} (y);
\end{tikzpicture}
\end{center}
\vskip-8pt
\begin{block}{Three conditions, stated carefully}
\begin{enumerate}\itemsep1pt
  \item \textbf{Relevance}: $\Cov(Z_i, D_i) \neq 0$. \alert{Testable.}
  \item \textbf{Independence}: $Z_i$ is as good as randomly assigned, no
        arrow into $Z$ from anything that also causes $Y$.
  \item \textbf{Exclusion}: $Z$ affects $Y$ \emph{only} through $D$. \alert{Not
        testable.}
\end{enumerate}
\end{block}
\vskip2pt
Conditions 2 and 3 are distinct and are often conflated. A randomized lottery
satisfies 2 by construction and can still violate 3.
\end{frame}

\begin{frame}{The Wald Estimator\core}
Take covariances of $Y_i = \alpha + \beta D_i + \epsilon_i$ with $Z_i$:
\begin{align*}
  \Cov(Z_i, Y_i) = \beta\,\Cov(Z_i,D_i) + \underbrace{\Cov(Z_i,\epsilon_i)}_{=\,0}
  \quad\Longrightarrow\quad
  \boxed{\;\beta_{IV} = \frac{\Cov(Z_i,Y_i)}{\Cov(Z_i,D_i)}\;}
\end{align*}
\pause
With a \emph{binary} instrument this is a ratio of two differences in means:
\begin{align*}
  \beta_{IV}
  = \frac{\E[Y_i\mid Z_i=1] - \E[Y_i\mid Z_i = 0]}
         {\E[D_i\mid Z_i=1] - \E[D_i\mid Z_i = 0]}
  = \frac{\text{reduced form}}{\text{first stage}} .
\end{align*}
\vskip2pt
Read it as: \emph{the effect of the instrument on the outcome, scaled up by how
much the instrument moved the treatment.}
\end{frame}

\begin{frame}{Intuition for the Ratio\core}
Divide numerator and denominator by $\Var(Z_i)$:
\begin{align*}
  \beta_{IV}
  = \frac{\Cov(Z_i,Y_i)/\Var(Z_i)}{\Cov(Z_i,D_i)/\Var(Z_i)}
  = \frac{\text{coefficient from regressing } Y \text{ on } Z}
         {\text{coefficient from regressing } D \text{ on } Z} .
\end{align*}
\begin{center}
\begin{tikzpicture}[scale=1.0,>=Stealth,thick,
  every node/.style={circle,draw=dukeblue,fill=white,inner sep=1pt,minimum size=8mm,font=\small}]
  \node (Z) at (0,0) {$Z$}; \node (D) at (2.4,0) {$D$}; \node (Y) at (4.8,0) {$Y$};
  \draw[->,dukeblue] (Z)--(D) node[draw=none,fill=none,midway,above]{\small$\gamma$};
  \draw[->,dukeblue] (D)--(Y) node[draw=none,fill=none,midway,above]{\small$\beta$};
  \draw[->,accent] (Z) to[bend right=30] node[draw=none,fill=none,midway,below,text=accent]{\small$\gamma\beta$} (Y);
\end{tikzpicture}
\end{center}
\vskip-6pt
\begin{itemize}\itemsep2pt
  \item The instrument's total effect on $Y$ travels along one path and equals
        $\gamma\beta$. The first stage recovers $\gamma$. Dividing isolates
        $\beta$.
  \item \alert{If $\gamma$ is small, we are dividing by something close to zero.}
        Hold that thought; it is the whole of the weak-instrument problem.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section[LATE]{What Does IV Estimate?}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}[allowframebreaks,t]{Four Types of People\core}
With a binary instrument and binary treatment, write $D_i(z)$ for the treatment
unit $i$ would take if assigned $Z_i = z$. Every unit is one of four types:
\begin{center}\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{lccl}
\toprule
Type & $D_i(0)$ & $D_i(1)$ & \\
\midrule
\textbf{Compliers} & 0 & 1 & do what the instrument says \\
Always-takers & 1 & 1 & treated regardless \\
Never-takers & 0 & 0 & untreated regardless \\
\alert{Defiers} & 1 & 0 & do the opposite \\
\bottomrule
\end{tabular}
\end{center}
\vskip4pt
\begin{itemize}\itemsep2pt
  \item We never observe anyone's type: we see $D_i(1)$ or $D_i(0)$, never both.
        Type is a latent variable, and this is the same fundamental problem as
        Week 5, one level up.
  \item Only \textbf{compliers} have their treatment moved by the instrument.
  \item \alert{Monotonicity} assumes there are no defiers: the instrument pushes
        everyone in the same direction, or not at all.
\end{itemize}
\end{frame}

\begin{frame}{Deriving the LATE: the Numerator}
Write the reduced form using the switching equation
$Y_i = Y_i\big(D_i(Z_i)\big)$ and split by type. Because $Z$ is independent of
potential outcomes and types,
\begin{align*}
  &\E[Y_i\mid Z_i=1] - \E[Y_i\mid Z_i=0] \\
  &\quad= \Pr(\text{AT})\underbrace{\big(\E[Y_i(1)\mid \text{AT}] - \E[Y_i(1)\mid \text{AT}]\big)}_{=\,0} \\
  &\quad\ + \Pr(\text{NT})\underbrace{\big(\E[Y_i(0)\mid \text{NT}] - \E[Y_i(0)\mid \text{NT}]\big)}_{=\,0} \\
  &\quad\ + \Pr(\text{C})\ \big(\E[Y_i(1)\mid \text{C}] - \E[Y_i(0)\mid \text{C}]\big) \\
  &\quad\ - \Pr(\text{D})\ \big(\E[Y_i(1)\mid \text{D}] - \E[Y_i(0)\mid \text{D}]\big) .
\end{align*}
\vskip2pt
Always- and never-takers take the \emph{same} treatment under both values of
$Z$, so \alert{by exclusion} their outcomes are unchanged and they contribute
zero. \alert{Monotonicity} kills the last line by assuming $\Pr(\text{D})=0$, 
note that defiers would enter with the \emph{wrong sign}.
\end{frame}

\begin{frame}{Deriving the LATE: the Denominator\core}
The first stage, $\E[D_i\mid Z_i=1] - \E[D_i\mid Z_i=0]$, recovers the complier share:
\begin{align*}
  \text{first stage}
  &= \Pr(\text{C}) + \Pr(\text{AT}) - \Pr(\text{AT}) - \Pr(\text{D})
     && \text{\footnotesize by type} \\
  &= \Pr(\text{C}) - \Pr(\text{D}) \\
  &= \Pr(\text{C})
     && \text{\footnotesize no defiers}
\end{align*}
\vskip3pt
Dividing:
\begin{block}{The LATE theorem \srcp{Imbens and Angrist 1994}}
\begin{align*}
  \frac{\E[Y\mid Z=1] - \E[Y\mid Z=0]}{\E[D\mid Z=1]-\E[D\mid Z=0]}
  = \E\big[\,Y_i(1) - Y_i(0)\ \big|\ \text{$i$ is a complier}\,\big] .
\end{align*}
\end{block}
\vskip3pt
\alert{IV does not estimate the ATE.} It estimates the average effect for the
subpopulation whose treatment status the instrument actually changed, a group
you cannot identify by name, defined by the instrument you happened to choose.
\end{frame}

\begin{frame}{Living With the LATE\core}
\begin{itemize}\itemsep3pt
  \item If effects are homogeneous, LATE $=$ ATE and none of this matters. That is
        rarely a defensible assumption in sociology.
  \item Different instruments for the same treatment estimate \emph{different}
        LATEs. When two IV papers disagree, that is not necessarily a
        contradiction; it may be two correct answers to two different questions.
  \item \textbf{You can describe the compliers even though you cannot name them}:
        their share is the first stage, and their covariate means follow because
        always-takers and never-takers are observed groups (Abadie's $\kappa$).
  \item \textbf{Compliers are often unusual}: for quarter-of-birth they are
        marginal dropouts; for judge-leniency designs they are defendants near the
        decision margin. Whether that group is the policy-relevant one is a
        substantive question you must answer, not a technicality.
\end{itemize}
\vskip2pt
\alert{Always characterize your compliers}: without that, a LATE is an average
effect for a group nobody can describe.
\end{frame}

\begin{frame}{Describing the Compliers \srcp{Abadie 2003}}
Binary $Z$ and $D$, no defiers. Among $Z = 0$ the treated are always-takers;
among $Z = 1$ the untreated are never-takers. So $\Pr(AT)$, $\Pr(NT)$,
$\E[X \mid AT]$, $\E[X \mid NT]$ are observed, $\Pr(C)$ is the first stage,
and the complier mean is what is left of the population mean:
\begin{align*}
  \E[X \mid C] = \frac{\E[X] - \Pr(AT)\,\E[X \mid AT] - \Pr(NT)\,\E[X \mid NT]}{\Pr(C)}
\end{align*}
\vskip-2pt
\textbf{Card}: $D$ = more than one year of college, $Z$ = near a four-year
college; $\Pr(AT) = 0.33$, $\Pr(NT) = 0.55$, $\Pr(C) = 0.12$ \srcp{figs/w7\_card.R}.
\vskip-3pt
\begin{center}\small
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}lcccc@{}}
\toprule
 & everyone & always-takers & never-takers & compliers \\
\midrule
share Black & 0.23 & 0.15 & 0.29 & 0.21 \\
share South & 0.40 & 0.50 & 0.38 & 0.26 \\
share metropolitan & 0.71 & 0.58 & 0.78 & 0.77 \\
\bottomrule
\end{tabular}
\end{center}
\vskip-2pt
Compliers are less Southern and more urban than always-takers. Dividing by
$\Pr(C) = 0.12$ amplifies noise: rough numbers.
\end{frame}

\begin{frame}{Can You Check Monotonicity? (1) What It Implies}
Not directly, $D_i(1)$ and $D_i(0)$ are never both seen; but ``no defiers'' can be \emph{refuted}.
\begin{itemize}\itemsep3pt
  \item \textbf{One sign everywhere.} If $Z$ only pushes $D$ up, then
        $\Pr(D=1 \mid Z=1, X=x) \ge \Pr(D=1 \mid Z=0, X=x)$ in every subgroup;
        a negative first stage somewhere is a rejection. Card \srcp{figs/w7\_card.R}:
\end{itemize}
\vskip-4pt
\begin{center}\small
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}lcccccc@{}}
\toprule
 & South & non-South & metro & non-metro & Black & other \\
\midrule
first stage & 0.31 & 0.34 & 0.37 & 0.27 & 0.28 & 0.35 \\
\bottomrule
\end{tabular}
\end{center}
\vskip-2pt
\begin{itemize}\itemsep3pt
  \item \textbf{Balke--Pearl inequalities} (binary $Z$, $D$). Switching $Z$ on
        can only \emph{add} treated units, so for any outcome set $A$,
        $\Pr(Y \in A, D=1 \mid Z=1) \ge \Pr(Y \in A, D=1 \mid Z=0)$ and
        $\Pr(Y \in A, D=0 \mid Z=0) \ge \Pr(Y \in A, D=0 \mid Z=1)$.
        \srct{Kitagawa (2015)} tests them; rejection means exclusion,
        independence, or monotonicity failed.
  \item Passing does not establish monotonicity; for lotteries and distance the
        case is institutional.
\end{itemize}
\end{frame}

\begin{frame}{Can You Check Monotonicity? (2) Bounding the Damage}
With defiers present and first stage $\Delta_D = \Pr(C) - \Pr(D)$:
\begin{align*}
  \text{Wald} &= \frac{\Pr(C)\,\tau_C - \Pr(D)\,\tau_D}{\Delta_D}
     && \text{\footnotesize numerator and denominator by type} \\
  \tau_C &= \frac{\text{Wald}\cdot\Delta_D + \Pr(D)\,\tau_D}{\Delta_D + \Pr(D)}
     && \text{\footnotesize solve for the complier effect}
\end{align*}
\vskip-2pt
Only $\Pr(D)$ and $\tau_D$ are unknown. Assume \alert{at most a share $\delta$
of defiers} and a bounded outcome, $|\tau_D| \le M$; the worst cases bound $\tau_C$.
\begin{itemize}\itemsep2pt
  \item \textbf{The judge toy example}: Wald $= -0.20$, $\Delta_D = 0.10$, $M = 1$,
        $\delta = 0.20$ gives $\tau_C \in [-0.73,\ 0.60]$: a first stage no larger
        than the defier share is fatal.
  \item \textbf{Sharp version} \srcp{Noack 2021}: sensitivity parameters for the
        defier share and the gap $\tau_D - \tau_C$; report how large they must be
        before the conclusion flips.
  \item \textbf{Rescue without a bound} \srcp{de Chaisemartin 2017}: if a subgroup
        of compliers matches the defiers in size and mean effect, Wald is the
        LATE of the remaining compliers. Untestable.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section[Strength]{Instrument Strength}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{Weak Instruments Are Worse Than No Instrument\core}
Recall $\hat\beta_{IV} = \Cov_n(Z,Y)/\Cov_n(Z,D)$. If the denominator
is near zero, small violations of exclusion get \emph{divided by} a small number:
\begin{align*}
  \plim \hat\beta_{IV} = \beta + \frac{\Cov(Z,\epsilon)}{\Cov(Z,D)} .
\end{align*}
\vskip3pt
\begin{itemize}\itemsep3pt
  \item An instrument that is one-tenth as endogenous as $D$ but one-hundredth as
        strongly correlated with it produces \alert{more} bias than plain OLS.
  \item The IV standard error is roughly the OLS standard error divided by
        $|\rho_{ZD}|$, the correlation between instrument and treatment: $\rho_{ZD} = 0.14$ costs a factor of 7.
  \item Worse, under weak identification $\hat\beta_{IV}$ has \alert{no Gaussian
        limit}: heavy tails, so the $t$-test fails no matter how large $n$ is.
  \item The problem \emph{worsens} with more instruments, because many weak
        instruments overfit the first stage (Bound--Jaeger--Baker, in the
        Angrist--Krueger frames).
\end{itemize}
\end{frame}

\begin{frame}{Diagnosing Instrument Strength\core}
\begin{itemize}\itemsep3pt
  \item \textbf{First-stage $F$} on the excluded instruments. The traditional rule
        of thumb is $F > 10$ \srcp{Staiger and Stock 1997; Stock and Yogo 2005},
        derived so that IV bias is under 10\% of OLS bias.
  \item \alert{That rule is about bias, not about test size, and it is now known
        to be far too lax for inference.}
        \srct{Lee, McCrary, Moreira, and Porter (2022)} show that a conventional
        $5\%$ $t$-test has correct size only when the first-stage $F$ exceeds
        roughly \textbf{104}. Their \texttt{tF} procedure replaces the $1.96$
        critical value with a larger one read off $F$: at $F = 10$ the adjusted
        critical value is about \textbf{3.43}, so the honest interval is roughly
        \textbf{1.75} times wider than the printed one.
  \item Use a \emph{robust} first-stage $F$ if you use robust or clustered
        standard errors in the second stage.
  \item With several endogenous regressors, use the \emph{Kleibergen--Paap} rank
        statistic; a strong $F$ on each instrument separately is not enough.
\end{itemize}
\end{frame}

\section[Applications]{Four Canonical Applications}

\begin{frame}{Four Canonical Applications}
\textbf{Goal}: read four classic instruments in one fixed order: the question,
the data, the instrument, the three conditions, the main results.
\vskip3pt
\begin{center}\scriptsize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}l>{\raggedright\arraybackslash}p{2.5cm}>{\raggedright\arraybackslash}p{2.7cm}>{\raggedright\arraybackslash}p{2.7cm}>{\raggedright\arraybackslash}p{2.6cm}@{}}
\toprule

%% ---- Angrist and Krueger 1991 -----------------------------------------------
\begin{frame}{In the Literature: Angrist and Krueger (1991)}
\textbf{The question.} What does a year of schooling ($D$) do to earnings
($Y$, log weekly wage)? Ability raises both, so OLS is biased up; reported
schooling is noisy, which pulls OLS down. The two biases have unknown
relative size.
\vskip3pt
\textbf{The data.} 1980 Census, 5\% sample: 329,509 men born 1930--39 with
positive earnings in 1979.
\vskip3pt
\textbf{The instrument.} $Z$ = born in the first quarter of the year.
Children start school in the calendar year they turn six but may leave at a
fixed \emph{age}, 16 or 17. A child born in January is older than a classmate
born in December in every grade and reaches the leaving age with less
schooling completed.
\end{frame}

\begin{frame}{Angrist and Krueger: the Three Conditions}
\begin{center}\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}lp{10.6cm}@{}}
\toprule
Relevance & First-quarter men have $0.109$ fewer years of schooling
  $(0.013)$. \alert{Testable, passes}, but the first stage is a tenth of a
  year. \\
Independence & Birth quarter is unrelated to earnings potential. Threat:
  season of birth varies with mothers' income and education and with
  health at birth, which also move earnings. \\
Exclusion & Birth quarter affects earnings only through years of schooling.
  Threat: the January child is the oldest in the class; relative age may
  shape test scores and confidence, and earnings, directly. \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}\itemsep2pt
  \item The two threats are different: the first puts an arrow \emph{into}
        $Z$, the second an arrow \emph{out of} $Z$ that bypasses $D$.
  \item \textbf{Whose effect.} The men moved by the instrument are those who
        would have left at the minimum age. The estimate is a return to
        schooling at the bottom of the distribution, the quantity compulsory
        attendance laws care about.
\end{itemize}
{\footnotesize \emph{Compliers} in the sense of the previous section: the people
whose schooling the instrument actually moved.}
\end{frame}

\begin{frame}{Angrist and Krueger: the Cautionary Half}
The paper is also the standard illustration of many weak instruments. It
combines instruments, which is \textbf{2SLS}, the section after the
applications, and shows the weak-identification problem at scale.
\vskip2pt
\begin{itemize}\itemsep2pt
  \item To gain precision, the authors interacted quarter of birth with state
        and year of birth: up to 180 instruments.
  \item \srct{Bound, Jaeger, and Baker (1995)}: with that many weak
        instruments, 2SLS is biased \emph{toward OLS}. Replacing quarter of
        birth with \emph{randomly generated} instruments gave similar
        estimates and standard errors.
  \item The mechanism is overfitting the first stage: $\hat D$ reproduces
        the endogenous variation we were trying to discard.
  \item \alert{More instruments is not more information.} Report the
        just-identified estimate beside any overidentified one.
\end{itemize}
Other instruments worth knowing: sibling sex composition for fertility,
lottery admission for school quality.
\end{frame}

\section[2SLS]{Two-Stage Least Squares}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{The Two Stages\core}
With covariates and possibly several instruments:
\begin{align*}
  \text{Stage 1:}\quad D_i &= \pi_0 + \pi_1 Z_i + \pi_2' W_i + v_i
  \quad\Longrightarrow\quad \hat D_i \\[3pt]
  \text{Stage 2:}\quad Y_i &= \alpha + \beta \hat D_i + \gamma' W_i + u_i
\end{align*}
\vskip3pt
\begin{itemize}\itemsep3pt
  \item $\hat D_i$ is the part of $D_i$ \emph{explained by the instrument}, which
        by assumption is uncorrelated with $\epsilon_i$. We regress on the clean
        part and discard the rest.
  \item \alert{The exogenous covariates $W_i$ must appear in \emph{both} stages.}
        Omitting them from the first stage is a common and serious error.
  \item \textbf{Order condition}: at least as many excluded instruments as
        endogenous regressors. \emph{Exactly identified} means equality;
        \emph{overidentified} means more.
  \item \alert{Never run the two stages by hand.} The second-stage standard errors
        would treat $\hat D$ as data rather than as an estimate. Use
        \texttt{ivreg()} or \texttt{feols(y \textasciitilde\ w | d \textasciitilde\ z)}.
\end{itemize}
\end{frame}

\begin{frame}{Why $W$ Goes in Both Stages}
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\begin{center}

\begin{frame}{IV Standard Error, Scalar Version (1)}
One $Y$, one $D$, one $Z$. Substitute the model for $Y_i$ into the Wald ratio:
\begin{align*}
  \hat\beta_{IV} = \frac{\Cov_n(Z_i, Y_i)}{\Cov_n(Z_i, D_i)}
    &= \frac{\Cov_n(Z_i,\ \alpha + \beta D_i + \epsilon_i)}{\Cov_n(Z_i, D_i)}
    && \text{\footnotesize substitute $Y_i = \alpha + \beta D_i + \epsilon_i$} \\
    &= \beta + \frac{\Cov_n(Z_i, \epsilon_i)}{\Cov_n(Z_i, D_i)}
    && \text{\footnotesize covariance is linear; the constant drops} \\
    &= \beta + \frac{\sum_i \tilde z_i\,\epsilon_i}{\sum_i \tilde z_i\,\tilde d_i}
    && \text{\footnotesize centered $Z$ and $D$; the $1/n$ cancel}
\end{align*}
\vskip-4pt
\begin{itemize}\itemsep1pt
  \item \textbf{The numerator is the random part}: independent terms
        $\tilde z_i \epsilon_i$ with mean zero and variance $\tilde z_i^{\,2}\sigma_i^2$; its
        variance is their sum.
  \item \textbf{The denominator is treated as fixed} (condition on $Z$ and $D$):
        the ``strong instrument'' step; the weak-IV section is about its failure.
\end{itemize}
\vskip-3pt
\begin{align*}
  \Var(\hat\beta_{IV} \mid Z, D) = \frac{\sum_i \tilde z_i^{\,2}\,\sigma_i^2}{\big(\sum_i \tilde z_i\,\tilde d_i\big)^2} .
\end{align*}
\end{frame}

\begin{frame}{IV Standard Error, Scalar Version (2)}
Take $\sigma_i^2 = \sigma^2$ for every $i$ and divide top and bottom by
$\sum_i \tilde z_i^{\,2}\sum_i \tilde d_i^{\,2}$; $\rho_{ZD}$ is the correlation of $Z$ and $D$:
\begin{align*}
  \Var(\hat\beta_{IV}) = \frac{\sigma^2 \sum_i \tilde z_i^{\,2}}{\big(\sum_i \tilde z_i\tilde d_i\big)^2}
  = \frac{\sigma^2}{\sum_i \tilde d_i^{\,2}}\cdot
    \frac{\sum_i \tilde z_i^{\,2}\sum_i \tilde d_i^{\,2}}{\big(\sum_i \tilde z_i\tilde d_i\big)^2}
  = \underbrace{\frac{\sigma^2}{n\,s_D^2}}_{\text{OLS variance}}\cdot\frac{1}{\rho_{ZD}^2}
\end{align*}
\begin{itemize}\itemsep3pt
  \item \alert{IV standard error $=$ OLS standard error $/\,|\rho_{ZD}|$}, for the
        same $\sigma$. The instrument explains a fraction $\rho_{ZD}^2$ of the
        variance in $D$, and only that fraction is used: $\rho_{ZD} = 0.14$
        costs a factor of 7.
  \item \textbf{Which residual.} $\sigma^2$ is estimated from
        $\hat\epsilon_i = Y_i - \hat\alpha - \hat\beta_{IV} D_i$, with the
        \emph{actual} $D_i$. A second stage run by hand uses $\hat D_i$ instead,
        gets a smaller residual, and prints too small a standard error.
  \item \textbf{Robust version.} Keep $\sigma_i^2$ and plug in $\hat\epsilon_i^{\,2}$:
        $\widehat{\Var} = \sum_i \tilde z_i^{\,2}\hat\epsilon_i^{\,2}/(\sum_i \tilde z_i\tilde d_i)^2$.
\end{itemize}
\end{frame}

Log wage on schooling, instrument near a four-year college, no controls
($n = 3{,}010$) \srcp{figs/w7\_card.R}.

\end{frame}

\begin{frame}{IV in Matrix Form\adv}
Start from the moment condition $\E[\bm Z' u] = \zero$ with $Y = D\beta_0 + u$.
The sample analogue $\bm Z'(\bm Y - \bm D\hat\beta) = \zero$ has no solution
when there are more instruments than regressors, so instead \emph{minimize} the
sample moments in a weighted norm:
\begin{align*}
  \hat\beta = \argmin_{b}\ \big(\bm Z'(\bm Y-\bm Db)\big)'
  (\bm Z'\bm Z)^{-1}\big(\bm Z'(\bm Y-\bm Db)\big) .
\end{align*}
The first-order condition gives
\begin{align*}
  \hat\beta = \Big(\bm D'\bm Z(\bm Z'\bm Z)^{-1}\bm Z'\bm D\Big)^{-1}
              \bm D'\bm Z(\bm Z'\bm Z)^{-1}\bm Z'\bm Y .
\end{align*}
\end{frame}

\begin{frame}{IV in Matrix Form: The 2SLS Estimator\adv}
Define the projection matrix $\bm P_Z = \bm Z(\bm Z'\bm Z)^{-1}\bm Z'$, the
same object as Week 1, now projecting onto the instrument space:
\begin{align*}
  \boxed{\;\hat\beta_{2SLS} = \big(\bm D'\bm P_Z \bm D\big)^{-1}\bm D'\bm P_Z\bm Y\;}
\end{align*}
Since $\bm P_Z\bm D$ \emph{is} the first-stage fitted value and $\bm P_Z$ is
idempotent, this is literally ``regress $\bm Y$ on $\hat{\bm D}$.'' And when
$\bm Z = \bm D$, $\bm P_Z\bm D = \bm D$ and 2SLS collapses to OLS: \alert{OLS is
the special case in which you claim everything is exogenous}.
\end{frame}

\begin{frame}{The IV Sandwich\adv}
Substitute $\bm Y = \bm D\beta_0 + \bm u$:
\begin{align*}
  \hat\beta - \beta_0 = \big(\bm D'\bm P_Z\bm D\big)^{-1}\bm D'\bm P_Z \bm u .
\end{align*}
Multiply by $\sqrt n$. Under regularity conditions and a \emph{strong}
instrument,
\begin{align*}
  \tfrac1n \bm D'\bm P_Z\bm D \pto \bm A \succ 0,
  \qquad
  \tfrac{1}{\sqrt n}\bm D'\bm P_Z\bm u \dto \mathcal N(\zero, \bm S) ,
\end{align*}
so by Slutsky $\sqrt n(\hat\beta-\beta_0)\dto \mathcal N(\zero, \bm A^{-1}\bm S\bm A^{-1})$.
\vskip3pt
The robust variance estimator is the familiar sandwich:
\begin{align*}
  \hat{V}(\hat\beta)
  = (\bm D'\bm P_Z\bm D)^{-1}\,\bm D'\bm P_Z\,
    \hat{V}(\bm u)\,\bm P_Z\bm D\,(\bm D'\bm P_Z\bm D)^{-1} ,
\end{align*}
plugging in $\hat{V}(\bm u) = \mathrm{diag}(\hat u_i^2)$, and clustering
exactly as in Week 2. \alert{Note the residuals $\hat u_i$ must be computed with
the actual $D_i$, not $\hat D_i$}; this is the other reason not to run the stages
by hand.
\end{frame}

\begin{frame}{Overidentification Tests: Read Them Carefully}
With more instruments than endogenous regressors, the Sargan--Hansen $J$ test
asks whether the instruments agree with one another.
\vskip3pt
\begin{itemize}\itemsep3pt
  \item \textbf{Rejection} says at least one instrument is invalid, \emph{or}
        that effects are heterogeneous, so the instruments estimate different
        LATEs and are supposed to disagree. It does not say which.
  \item \textbf{Non-rejection is nearly worthless.} It is consistent with all the
        instruments being invalid \emph{in the same direction}, and the test has
        little power.
  \item The $J$ test cannot validate exclusion. Nothing can. The defence is
        substantive: tell a story about why $Z$ could not affect $Y$ except
        through $D$, and look for \emph{placebo outcomes} it should not move and
        \emph{placebo samples} in which the first stage should be absent.
\end{itemize}
\end{frame}

\section[Weak IV]{Inference Under Weak Instruments}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{Where $\E[(Y - \beta D)\,Z] = 0$ Comes From}
The IV assumption: the instrument is unrelated to the error, $\Cov(Z_i, \epsilon_i) = 0$.
Substitute what the error is:
\begin{align*}
  \epsilon_i &= Y_i - \alpha - \beta D_i
     && \text{\footnotesize the structural equation, solved for $\epsilon_i$} \\
  0 = \Cov(Z_i, \epsilon_i) &= \Cov(Z_i,\ Y_i - \beta D_i)
     && \text{\footnotesize $\alpha$ is a constant} \\
     &= \Cov(Z_i, Y_i) - \beta\,\Cov(Z_i, D_i)
     && \text{\footnotesize covariance is linear}
\end{align*}
\vskip4pt
\begin{itemize}\itemsep4pt
  \item \textbf{In words}: strip the treatment's part, $\beta D$, out of $Y$;
        what remains must be unrelated to $Z$. The true $\beta$ satisfies this
        equation.
  \item \textbf{Solve it for $\beta$} and you get the Wald estimator,
        $\Cov(Z,Y)/\Cov(Z,D)$. Nothing new is assumed.
  \item $\E[(Y_i - \beta D_i)\,\tilde Z_i]$ with $\tilde Z_i = Z_i - \E[Z_i]$ is the
        same covariance; the tilde only centers $Z$.
\end{itemize}
\end{frame}

\begin{frame}{Two Ways to Use the Same Equation}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Solve it} (Wald, 2SLS): $\hat\beta_{IV} = \Cov_n(Z,Y)/\Cov_n(Z,D)$.
Divides by the first stage; unreliable when that is small and noisy.
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Check it} (Anderson--Rubin): guess $\beta_0$, ask whether
$\Cov_n(Z,\ Y - \beta_0 D) = 0$. Never divides.
\end{column}
\end{columns}
\vskip6pt
\textbf{Card's data} ($Z$ = near a college, $D$ = schooling, $Y$ = log wage):
$\Cov_n(Z,Y) = 0.0338$, $\Cov_n(Z,D) = 0.1798$ \srcp{figs/w7\_weakiv.R}.
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lcccccc@{}}
\toprule
guess $\beta_0$ & 0 & 0.10 & 0.144 & 0.188 & 0.25 & 0.30 \\
\midrule
$\Cov_n(Z, Y - \beta_0 D) = 0.0338 - \beta_0 \times 0.1798$ & 0.034 & 0.016 & 0.008 & 0.000 & $-0.011$ & $-0.020$ \\
\bottomrule
\end{tabular}
\end{center}
\vskip4pt
Zero at $\beta_0 = 0.188$, the Wald estimate. For the other guesses: how far from
zero is too far? That needs a scale.
\end{frame}

\begin{frame}{The AR Statistic, Step by Step}
\begin{enumerate}\itemsep4pt
  \item Guess $\beta_0$; build $U_i = Y_i - \beta_0 D_i$.
  \item Regress $U_i$ on $Z_i$. Since $U_i = \alpha + (\beta - \beta_0) D_i + \epsilon_i$,
        the true slope is $(\beta - \beta_0)\,\pi$: zero if the guess is right.
  \item $C(\beta_0) = t^2$ of that slope; $\chi^2_1$ under $H_0\!: \beta = \beta_0$.
        Reject the guess if $C(\beta_0) > 3.84$.
\end{enumerate}
\vskip2pt
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lcccccc@{}}
\toprule
guess $\beta_0$ & 0 & 0.10 & 0.144 & 0.188 & 0.25 & 0.30 \\
\midrule
slope of $U$ on $Z$ & 0.156 & 0.073 & 0.037 & 0.000 & $-0.051$ & $-0.093$ \\
$t$ & 9.10 & 4.25 & 1.91 & 0.00 & $-1.94$ & $-3.02$ \\
$C(\beta_0) = t^2$ & 82.7 & 18.0 & 3.66 & 0.00 & 3.76 & 9.12 \\
verdict & reject & reject & keep & keep & keep & reject \\
\bottomrule
\end{tabular}
\end{center}
\vskip4pt
Step 2 is an ordinary regression $t$-test: $\hat\pi$ never appears in a
denominator. A weak instrument shrinks $(\beta - \beta_0)\pi$ for every guess,
so fewer guesses are rejected: a wider set, not a wrong one.
\end{frame}

\begin{frame}{Inverting the Test: the AR Confidence Region}
Keep every guess the data cannot reject, $CR(\beta) = \{\beta_0 : C(\beta_0) \le c_{1-a}\}$.
\textbf{``Works under a weak instrument''} means: covers the truth 95\% of the
time whatever $F$ is. Simulation, $n = 500$, $\beta = 1$, 4,000 draws \srcp{figs/w7\_weakiv.R}:
\vskip-3pt
\begin{center}\small
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}lcccc@{}}
\toprule
median first-stage $F$ & 1.8 & 6.0 & 9.7 & 101 \\
\midrule
Wald interval covers the truth & 87\% & 91\% & 92\% & 95\% \\

\begin{itemize}\itemsep1pt
  \item \textbf{Why Wald fails}: its standard error is too \emph{small}, not too
        large; $\hat\beta_{IV}$ leans toward OLS with heavy tails, and the printed
        normal interval is not there.
  \item \textbf{What AR does}: at $F \approx 6$, a bounded interval in 69\% of
        draws, wider than Wald's, else an unbounded region: it admits what the
        data cannot pin down, and coverage stays at 95\%.
\end{itemize}
\end{frame}

\section[Judge IV]{Judge and Examiner Designs}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{The Design}
\begin{itemize}\itemsep1pt
  \item \textbf{Goal}: among people \emph{convicted of a felony}, the effect of a
        prison sentence ($D = 1$) rather than probation ($D = 0$) on later
        reoffending or employment $Y$. The pool is the convicted, not the public.
  \item \textbf{Problem}: within that pool, judges send the worse records and
        graver offenses to prison, and the same things predict $Y$; prisoners and
        probationers are not comparable.
  \item \textbf{Variation}: cases are assigned to judges at random within a
        court and year, and judges differ persistently in how often they
        imprison. $Z_i$ = the leniency of the judge defendant $i$ drew.
\end{itemize}
\vskip-19pt
\begin{center}
\begin{tikzpicture}[scale=0.74,>=Stealth,
  every node/.style={circle,draw=dukeblue,fill=white,inner sep=1pt,minimum size=6mm,font=\scriptsize}]

  \node[draw=gray,dashed] (u) at (3.6,1.3) {$U$};

  \draw[->,accent,dashed,thick] (z) to[bend right=28] node[rectangle,draw=none,fill=none,midway,below,font=\scriptsize,text=accent]{sentence length, fines, probation terms} (y);
  \node[rectangle,draw=none,fill=none,font=\scriptsize,text=gray,anchor=west] at (4.2,1.3) {record, dangerousness};
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}{Constructing the Instrument}
Leave-one-out leniency: your judge's incarceration rate in everyone else's
cases (within court and year), $Z_i = \frac{1}{n_{j(i)} - 1}\sum_{k \neq i:\ j(k) = j(i)} D_k$.
\begin{itemize}\itemsep0pt
  \item \textbf{Why leave your own case out.} The leave-in mean contains
        $D_i$, which is correlated with $\epsilon_i$; the instrument inherits
        $\Cov(D_i,\epsilon_i)/n_j$ of it, shrinking with cases per judge only.
  \item \textbf{Judge dummies are the leave-in mean in disguise.} With $J - 1$
        judge indicators as instruments (Harding et al.; Kling 2006) the fitted
        value is the judge's rate over \emph{all} cases, own case included: same bias.
\end{itemize}
\vskip1pt
\textbf{Simulation} \srcp{figs/w7\_judge.R}: 100 judges, a confounder raising
$D$ and $Y$, true $\tau = 0.20$, 500 replications.

\end{frame}

\begin{frame}{The Three Conditions, Judge Version}
\begin{itemize}\itemsep3pt
  \item \textbf{Relevance.} Judges differ and courts have thousands of cases;
        $F$ is rarely the problem. Compute $Z$ and $F$ within court $\times$ year.
  \item \textbf{Independence.} Random assignment is an institutional fact and
        partly testable: regress $Z$ on defendant and case characteristics
        (race, age, priors, offense); nothing should predict the draw.
        Michigan assigns by a random number generator \srcp{Harding et al.\ 2019}.
  \item \textbf{Exclusion.} The harsh judge must change $Y$ only through
        prison versus probation. The same judge also sets sentence length,
        fines, and probation conditions, and speaks differently from the
        bench: each is a path from $Z$ to $Y$. Remedies: instrument every
        dimension of the sentence at once (Harding et al.: prison, jail, jail
        with probation, three sentence lengths), or define the treatment as
        the whole sentence package.
  \item \textbf{Monotonicity.} With two judges it reads ``no defiers.'' With
        many judges it must hold for every pair of judges and every defendant.
\end{itemize}
\end{frame}

\begin{frame}{Monotonicity With Many Judges: a Worked Failure}
Two judges, two case types in equal shares.
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lccc@{}}
\toprule
share incarcerated & drug cases (50\%) & property cases (50\%) & all cases \\
\midrule
Judge A & 0.90 & 0.30 & 0.60 \\
Judge B & 0.50 & 0.50 & 0.50 \\
\bottomrule
\end{tabular}
\end{center}
A is harsher overall, so $Z = 1$ for A. Among drug cases A creates 0.40
compliers; among property cases A is the lenient one and creates 0.20
\alert{defiers}.
\begin{align*}
  \text{first stage} &= 0.5(0.40) - 0.5(0.20) = 0.10
     && \text{\footnotesize defiers subtract} \\
  \text{reduced form} &= 0.5(0.40)\,\tau_{\text{drug}} - 0.5(0.20)\,\tau_{\text{prop}}
     && \text{\footnotesize numerator of the LATE derivation} \\
  \text{Wald} &= 2\,\tau_{\text{drug}} - \tau_{\text{prop}}
     && \text{\footnotesize weights sum to one; one is negative}
\end{align*}
With $\tau_{\text{drug}} = 0.10$ and $\tau_{\text{prop}} = 0.40$ the Wald
estimand is $-0.20$: every effect is positive, the IV estimand is negative.
Within each case type the ordering is stable and the Wald ratio recovers that
type's $\tau$.
\end{frame}

\begin{frame}{Detecting a Monotonicity Failure}
If judge A is harsher than B for one kind of defendant, A must be harsher for
every kind: plot each judge's rate on one case type against the other
\srcp{Dobbie, Goldin, and Yang 2018; figs/w7\_mono.R}.
\vskip-7pt
\begin{center}
\includegraphics[width=0.5\linewidth]{figs/mono_judges.pdf}
\end{center}
\vskip-5pt
\begin{itemize}\itemsep1pt
  \item \textbf{Holds}: judges line up alike on both types; correlation of the
        two rates $0.92$, 88\% of judge pairs ordered the same way.
  \item \textbf{Fails}: harsh on drugs means lenient on property; correlation

\begin{frame}{Addressing a Monotonicity Failure}
\begin{itemize}\itemsep4pt
  \item \textbf{The common fix: build leniency within strata.} Compute $Z_i$
        from the judge's other cases of the \emph{same type} as case $i$ (case
        type $\times$ court $\times$ year), and let the first stage differ by
        stratum. Within a stratum judges are consistently ordered, the Wald
        ratio recovers that stratum's $\tau$, and 2SLS pools the strata with
        positive weights.
  \item \textbf{The common assumption: average monotonicity}
        \srcp{Mogstad, Torgovitsky, and Walters 2021}. 2SLS with a leniency
        instrument is a positively weighted average of effects if, for every
        defendant, leniency and that defendant's own treatment covary
        non-negatively across judges. Judges may disagree on some cases; no
        defendant's treatment may run against leniency on average. Plausible
        when judge differences are mostly a common harshness, as in panel A.
  \item \textbf{A formal test} \srcp{Frandsen, Lefgren, and Leslie 2023}: with a
        bounded outcome, exclusion and monotonicity bound the slope of the
        per-judge mean outcome against the per-judge incarceration rate by the
        range of $Y$. \texttt{ivcheck::iv\_testjfe} in R.
\end{itemize}
\end{frame}

\begin{frame}{Whose Effect?}
\begin{itemize}\itemsep3pt
  \item \textbf{Compliers are the marginal defendants}: those whose sentence
        depends on which judge they drew. Not the mandatory minimums, not
        first-degree murder, not the obvious probation cases. In the toy
        example the pairwise complier share is the first stage, 0.10.
  \item With a continuous $Z$, 2SLS averages pairwise LATEs across the leniency
        distribution, more weight where judges are dense.
  \item \textbf{Describe them.} Complier means of covariates are recoverable
        by Abadie's $\kappa$ reweighting; judge papers report a table of
        complier characteristics next to the full sample.
  \item \textbf{For once the LATE is the policy parameter.} A sentencing
        reform moves exactly the marginal cases, not the always-incarcerated.
  \item \textbf{Different margins, different compliers.} Prison versus
        probation at sentencing \srcp{Harding et al.\ 2017} and detention
        versus release before trial \srcp{Dobbie, Goldin, and Yang 2018} are
        different populations; their LATEs need not agree.
\end{itemize}
\end{frame}

\begin{frame}{In the Literature: Harding, Morenoff, Nguyen, and Bushway}
\textbf{Data}: everyone sentenced for a felony in Michigan, 2003--2006
($n = 111{,}110$), administrative records to 2013--2015; cases with no
judicial discretion dropped. \textbf{Instruments}: judge dummies and judge
$\times$ presentence-characteristic interactions in 2SLS, six first stages;
LIML as a check.
\vskip2pt
\begin{itemize}\itemsep2pt
  \item \textbf{PNAS 2017.} Prison rather than probation raises the probability
        of being back in prison within five years by about 20 points for
        nonwhite and white defendants alike; technical parole violations
        account for 18 and 14 of those points, new felony sentences for 4 and
        9. No effect on new felony convictions after release.
  \item \textbf{AJS 2018.} The employment effect runs mainly through
        incapacitation, the sentence and the later returns; post-release
        penalties are concentrated among whites with a presentence work history.
  \item \textbf{Nature Human Behaviour 2019.} Violent arrests fall by 8.4
        points over five years when time inside is counted; after release, no
        detectable effect on violence.
\end{itemize}

\begin{frame}{Reading Harding et al.\ with This Week's Tools}
\begin{itemize}\itemsep3pt
  \item \textbf{Many instruments.} Judge dummies times characteristics run to
        hundreds; 2SLS leans toward OLS, hence the LIML check. One leave-one-out
        leniency per endogenous variable is now the default \srcp{Dobbie, Goldin,
        and Yang 2018; Bhuller et al.\ 2020}.
  \item \textbf{Exclusion} is handled by widening the treatment: six sentence
        dimensions instrumented together.
  \item \textbf{Monotonicity} is assumed, not tested.
  \item \textbf{The estimand} is the LATE for marginal felony defendants, the

\end{itemize}

defendants on nine random calendars, no effect on re-arrest.
\srct{Loeffler (2013)}: 20,000 Cook County felony defendants, no effects on
recidivism or employment. In economics, \srct{Kling (2006)},

\section[Shift-share]{Shift-Share Instruments}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}\sectionpage\end{frame}

\begin{frame}{The Problem and the Instrument}
\textbf{Goal}: the effect of a local shock $x_\ell$ (job loss, immigration,
import competition) on a local outcome $y_\ell$ (marriage, mortality, votes)
when $x_\ell$ responds to local conditions that also move $y_\ell$.
\vskip2pt
\textbf{Idea} \srcp{Bartik 1991; Card 2001}: predict the local shock from
national shocks by industry, weighted by baseline exposure,
\begin{align*}
  z_\ell = \sum_{k=1}^{K} s_{\ell k}\, g_k ,
\end{align*}
$s_{\ell k}$ the baseline share of industry $k$ in place $\ell$, $g_k$ the
national change in $k$. Card (2001): past settlement shares by origin country,
times national inflows by origin.
\textbf{Plan}
\begin{enumerate}\itemsep0pt
  \item Three places and two industries, by hand.
  \item Two routes to validity: exogenous shares or exogenous shocks.
  \item Proof by simulation, then inference.
  \item Autor, Dorn, and Hanson (2019) on marriage.
\end{enumerate}
\end{frame}

\begin{frame}{Three Places, Two Industries, by Hand}
National changes over the decade: manufacturing $g_M = -10\%$, services
$g_S = +5\%$. Baseline employment shares:
\begin{center}\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}lccl@{}}
\toprule
place & $s_{\ell M}$ & $s_{\ell S}$ & $z_\ell = s_{\ell M} g_M + s_{\ell S} g_S$ \\
\midrule
A & 0.6 & 0.4 & $0.6(-10) + 0.4(5) = -4$ \\
B & 0.2 & 0.8 & $0.2(-10) + 0.8(5) = +2$ \\
C & 0.4 & 0.6 & $0.4(-10) + 0.6(5) = -1$ \\
\bottomrule
\end{tabular}
\end{center}
\vskip2pt
\begin{itemize}\itemsep2pt
  \item The shocks are common to every place; $z_\ell$ varies only through the
        shares. The design compares A with B: the place that was manufacturing
        when manufacturing collapsed.
  \item Estimation is 2SLS of $y_\ell$ on $x_\ell$ with $z_\ell$ as the one
        excluded instrument, controls in both stages. $K$ industries are not
        $K$ instruments; Bound--Jaeger--Baker does not apply. What $K$ changes
        is the inference.
\end{itemize}
\end{frame}

\begin{frame}{Two Routes to Validity}
\begin{center}\scriptsize
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}p{2.4cm}p{5.1cm}p{5.1cm}@{}}
\toprule
 & shares exogenous \srcp{Goldsmith-Pinkham, Sorkin, and Swift 2020} & shocks exogenous \srcp{Borusyak, Hull, and Jaravel 2022} \\
\midrule
as good as random & the baseline shares $s_{\ell k}$, given controls & the shocks $g_k$ across industries; shares may be endogenous \\
the other piece & shocks only set the weights; the instrument is really the $K$ share variables & shares only set the exposure; the shocks are the identifying variation \\
reads like & a difference-in-differences with continuous exposure: high-share places would have trended like low-share ones & an industry-level IV with $K$ observations weighted by average exposure $\bar s_k$ \\
typical setting & Card enclaves; few, large shocks & the China shock; many industries, idiosyncratic shocks \\
what to check & pre-trends by exposure; Rotemberg weights, which shares carry the estimate & shock balance, do the $g_k$ predict industry averages of covariates; effective number of shocks $1/\sum_k \bar s_k^{\,2}$ \\
\bottomrule
\end{tabular}
\end{center}
\vskip4pt
Choose the story that fits the setting and defend that one; the two require
different checks.
\end{frame}

\begin{frame}{Proof by Simulation}
300 places, 80 industries, true $\beta = 1$, 1,000 replications, first-stage
$R^2 \approx 0.20$. A local supply shock enters both $x_\ell$ and $y_\ell$, so
OLS is biased \srcp{figs/w7\_shiftshare.R}.
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\includegraphics[width=\linewidth]{figs/shiftshare_sim.pdf}
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}lcc@{}}
\toprule
scenario & OLS & SSIV \\
\midrule
A. both stories hold & 1.65 & 0.99 \\
B. shares endogenous & 1.65 & 0.98 \\
C. shocks not random & 1.69 & \alert{1.35} \\
\bottomrule
\end{tabular}
\end{column}
\end{columns}
\vskip2pt
\begin{itemize}\itemsep1pt
  \item \textbf{B} (shocks random) is the Borusyak--Hull--Jaravel point: an
        endogenous share does no harm, because $z_\ell$ averages many mean-zero shocks.
  \item \textbf{C} is the failure neither story survives: a national shock
        that moves $y$ through a channel other than $x$ (the industry's
        technology, unions, politics).
\end{itemize}
\end{frame}

\begin{frame}{Inference: Places That Look Alike Are Not Independent}
\srct{Ad\~ao, Kolesár, and Morales (2019)}: if $y_\ell$ carries an industry
component $\sum_k s_{\ell k}\nu_k$, places with similar shares have correlated
residuals wherever they sit on the map. Robust and state-clustered standard
errors miss this and over-reject. Coverage of 95\% intervals in the simulation
\srcp{figs/w7\_shiftshare.R}:
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lcc@{}}
\toprule
 & robust SE & shock-level SE \\
\midrule
D. valid, industry-correlated errors & 0.90 & 0.94 \\
C. shocks not random (biased) & 0.07 & 0.13 \\
\bottomrule
\end{tabular}
\end{center}
\vskip2pt
\begin{itemize}\itemsep2pt
  \item \textbf{Shock-level regression} \srcp{Borusyak, Hull, and Jaravel 2022}:
        average $y_\ell$ and $x_\ell$ to industries with exposure weights,
        instrument with $g_k$, weight by $\bar s_k$. Same point estimate as the
        place-level 2SLS; the standard error treats the $K$ shocks as the
        independent units (\texttt{ShiftShareSE} implements Ad\~ao et al.).
  \item Few effective shocks behave like a small sample: report
        $1/\sum_k \bar s_k^{\,2}$ next to the first-stage $F$.
\end{itemize}
\end{frame}

\begin{frame}{In the Literature: Autor, Dorn, and Hanson (2019)}
The title is borrowed from \srct{Wilson (1996)}: does the collapse of
manufacturing lower young men's value in the marriage market?
\vskip3pt
\begin{itemize}\itemsep3pt
  \item \textbf{Shock}: growth in Chinese import penetration per worker in
        commuting zone $\ell$, $x_\ell = \sum_k s_{\ell k}\,\Delta M_k / L_k$,
        with male and female employment shares separately.
  \item \textbf{Instrument}: the same sum with China's export growth to eight
        other high-income countries and lagged shares \srcp{Autor, Dorn, and
        Hanson 2013}: the exclusion claim is about shocks, Chinese productivity
        rather than U.S.\ demand. 722 commuting zones, 1990--2000 and
        2000--2014 stacked, standard errors clustered by state.
  \item \textbf{Results}: a unit shock, the average decade-level rise in
        exposure, lowers the share of women aged 18--39 currently married by
        0.95 points and raises the never-married share by 1.2. Shocks to
        male-intensive industries do the work.
\end{itemize}
Shift-share designs are only now arriving in sociology journals; the first
China-shock paper in one is next.
\end{frame}

\begin{frame}{In the Literature: Shen (2026)}
\srct{Shen (2026)}, \textit{Sociological Science}: did the deterioration of men's
labor markets raise the correlation between spouses' earnings? The usual story
credits wives' rising employment; this one asks about husbands' decline.
\vskip3pt
\begin{itemize}\itemsep3pt
  \item \textbf{Data.} Census and ACS, 722 commuting zones, two stacked decadal
        changes (1990--2000, 2000--2014), $n = 1{,}444$. Outcome: the change in
        the zone's spousal earnings correlation.
  \item \textbf{Two shocks.} The Autor--Dorn--Hanson exposure
        $\sum_j (L_{ij}/L_i)\,\Delta Z^{u}_{ij}$ is split by each industry's male
        and female employment share in the zone, giving a male-specific and a
        female-specific import shock (mining versus textiles).
  \item \textbf{Two instruments.} The same sums built from China's export
        growth to eight other high-income countries and 1980 industry shares.
        Two endogenous regressors, so strength is reported as the
        Sanderson--Windmeijer conditional $F$ (113 and 124) and the
        Kleibergen--Paap $F$ (62). Standard errors clustered by state.
\end{itemize}
\end{frame}

\begin{itemize}\itemsep3pt
  \item \textbf{Mechanism.} A unit male shock adds 121 (33) annual work hours

        (27) at the bottom: families adapt to men's decline unequally, and the

  \item \textbf{Exclusion is the shocks story}: the non-U.S. component of
        Chinese export growth is as good as random with respect to local
        determinants of the correlation, given controls \srcp{Borusyak, Hull,
        and Jaravel 2022}.
  \item \textbf{Two threats the paper names.} Women are scarce in tradable
        manufacturing, so the female shock measures women's conditions poorly;
        and trade shocks also change who marries \srcp{Autor, Dorn, and Hanson 2019}.
\end{itemize}
\end{frame}

\begin{frame}[allowframebreaks,t]{A Reader's Checklist\core}
Adapted from \srct{Sovey and Green (2011)}. Any IV paper should report:
\begin{enumerate}\itemsep2pt
  \item The instrument, and an explicit argument for exclusion, not just a
        citation to a prior paper that used it.
  \item The \textbf{first stage}, its coefficient, and its robust $F$.
  \item The \textbf{reduced form} ($Y$ on $Z$), with its estimate and interval.
        It needs no instrument-strength assumption, so it is the most robust piece
        of evidence in the design, and if it is indistinguishable from zero the
        IV estimate rests on very little.
  \item The \textbf{OLS} estimate alongside, so the reader can see the direction
        and size of the correction, and whether it goes the way theory predicts.
  \item Who the \textbf{compliers} are, and their share.
  \item Weak-instrument-robust inference if $F$ is not comfortably large.
  \item The just-identified estimate if the paper is overidentified.
  \item Placebo checks: outcomes and subgroups $Z$ should not affect.
\end{enumerate}
\vskip2pt
\alert{If a paper reports only the 2SLS coefficient, that is a red flag.}
\end{frame}

\begin{frame}[allowframebreaks,t]{When IV Is the Wrong Tool\core}
\begin{itemize}\itemsep3pt
  \item \textbf{No credible instrument exists.} This is the normal state of
        affairs. An implausible instrument is worse than an honest
        selection-on-observables analysis with sensitivity bounds.
  \item \textbf{The LATE is not the quantity of interest.} If your policy question
        concerns always-takers, the compliers' effect does not answer it.
  \item \textbf{Interpretation drifts.} Papers routinely estimate a LATE and then
        discuss it as if it were an ATE.
  \item \textbf{Monotonicity is doing hidden work.} In judge-leniency and
        examiner designs, it must hold for \emph{every} judge simultaneously, 
        a much stronger claim than it first appears.
\end{itemize}
\vskip3pt
\begin{block}{The design-based cousins}
When the instrument is a discontinuity in an assignment rule, you have
\textbf{RDD} (Week 9), and indeed fuzzy RD \emph{is} local IV. When it is a
policy change over time, you have \textbf{DID} (Week 10).
\end{block}
\end{frame}

\begin{frame}{Looking Ahead}
\begin{itemize}\itemsep4pt
  \item IV replaces ``I measured all confounders'' with ``I found variation that
        is as good as random.'' Both are assumptions; the second is easier to
        argue about substantively, and easier for readers to attack, which is
        a virtue.
  \item The price is the estimand: LATE, for compliers you cannot name, defined
        by the instrument you chose.
  \item Notice how much identification work was done by \emph{functional form} in
        AJR, ``linear in distance from the equator, plus continent dummies.''
        Week 13 asks briefly what can be done when you are unwilling to assume
        that.
\end{itemize}
