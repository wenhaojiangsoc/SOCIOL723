\begin{frame}{When There Is Only One Treated Unit\core}
Difference-in-differences needs a control \emph{group}. What if the treated unit
is California, or East Germany, or a single firm?
\vskip4pt
\begin{itemize}\itemsep3pt
  \item Comparative case studies have always done this informally: ``compare
        California to the rest of the country.'' The choice of comparison was
        made by the author, and the reader had to trust it.
  \item \alert{Synthetic control} makes that choice a data-driven, transparent,
        pre-registered procedure: construct a weighted average of untreated units
        that reproduces the treated unit's pre-treatment trajectory.
  \item ``Arguably the most important innovation in the policy evaluation
        literature in the last 15 years'' \srcp{Athey and Imbens 2017}.
\end{itemize}
\vskip4pt
The identifying idea is not new. What is new is that the weights are chosen by an
algorithm, reported in a table, and checked against placebo units.
\end{frame}

\section[Synthetic Control]{The Synthetic Control Estimator}

\begin{frame}\sectionpage\end{frame}

\begin{frame}{Setup\core}
Unit 1 (California) is treated from year $T_0+1$ onward. Units $j = 2,\dots,J+1$,
the \alert{donor pool}, are never treated. For each year $t > T_0$ the effect is
\begin{align*}
  \tau_t = Y_{1t}(1) - Y_{1t}(0) .
\end{align*}
We observe $Y_{1t}(1)$; the whole problem is the number $Y_{1t}(0)$: California's
cigarette sales in, say, 1995 had Prop 99 never passed.
\vskip4pt
\begin{block}{Every method today has the same form}
Predict $Y_{1t}(0)$ by a weighted sum of the donors' outcomes in the same year,
\begin{align*}
  \hat Y_{1t}(0) = w_2 Y_{2t} + w_3 Y_{3t} + \dots + w_{J+1} Y_{J+1,t}
  = \textstyle\sum_{j=2}^{J+1} w_j Y_{jt} ,
\end{align*}
and estimate $\hat\tau_t = Y_{1t} - \hat Y_{1t}(0)$. Methods differ only in how
they choose the weights $w_j$.
\end{block}
\end{frame}

\begin{frame}{Running Example: Proposition 99}
California raised its cigarette tax by 25 cents a pack in 1989 and funded
anti-smoking campaigns. Outcome: packs sold per capita. Donors: 38 states with
no large tobacco program or tax increase in the period.
\vskip-2pt
\begin{center}
\includegraphics[width=0.72\linewidth]{figs/prop99_naive.pdf}
\end{center}
\vskip-9pt
\begin{itemize}\itemsep1pt
  \item California falls faster than the donor average, and it was already
        doing so before 1989 (1988: 90.1 against 113.8) \srcp{figs/w11\_scalar.R}.
  \item How much of the post-1989 gap is the law? Start with the naive answer.
\end{itemize}
\end{frame}

\begin{frame}{The Naive Approach: the Donor Average\core}
Give every donor the same weight, $w_j = 1/38$, and add California's average
pre-1989 gap: this is DiD (Week 10) \srcp{figs/w11\_scalar.R}.
\vskip3pt
\begin{center}\small
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}lccc@{}}
\toprule
 & 1970--88 mean & 1989--2000 mean & change \\
\midrule
California & 116.2 & 60.4 & $-55.8$ \\
average of 38 donors & 130.6 & 102.1 & $-28.5$ \\
\midrule
difference & $-14.4$ & $-41.7$ & $\mathbf{-27.3}$ \\
\bottomrule
\end{tabular}
\end{center}
\vskip3pt
\begin{itemize}\itemsep2pt
  \item DiD assumes California would have fallen as fast as the donor average.
  \item It was not doing so before the law: from 1970 to 1988 California fell by
        32.9 packs, the donor average by 6.3. Parallel trends fails in the
        pre-period, so $-27.3$ is too large in absolute value.
  \item The fix: choose the weights so that the donors' weighted past
        \emph{looks like} California's past.
\end{itemize}
\end{frame}

\begin{frame}{Two Donors, One Weight\core}
Start with two donors, Utah and Nevada. One number to choose, $w$:
\begin{align*}
  \hat Y_{\text{CA},t}(0) = w\, Y_{\text{UT},t} + (1-w)\, Y_{\text{NV},t} ,
  \qquad 0 \leq w \leq 1 .
\end{align*}
Pick $w$ to make the pre-period gaps as small as possible:
{\small
\begin{align*}
  \text{SSR}(w)
  &= \textstyle\sum_{t=1970}^{1988}\big(Y_{\text{CA},t} - w\,Y_{\text{UT},t} - (1-w)\,Y_{\text{NV},t}\big)^2
  && \text{\footnotesize squared pre-period gaps} \\
  &= \textstyle\sum_{t}\big[(Y_{\text{CA},t} - Y_{\text{NV},t}) - w\,(Y_{\text{UT},t} - Y_{\text{NV},t})\big]^2
  && \text{\footnotesize regroup the $w$ terms} \\
  0 &= -2\textstyle\sum_{t}(Y_{\text{UT},t} - Y_{\text{NV},t})\big[(Y_{\text{CA},t} - Y_{\text{NV},t}) - w\,(Y_{\text{UT},t} - Y_{\text{NV},t})\big]
  && \text{\footnotesize derivative set to 0} \\
  w &= \frac{\sum_t (Y_{\text{UT},t} - Y_{\text{NV},t})(Y_{\text{CA},t} - Y_{\text{NV},t})}{\sum_t (Y_{\text{UT},t} - Y_{\text{NV},t})^2}
  && \text{\footnotesize solve for $w$}
\end{align*}}
\begin{itemize}\itemsep2pt
  \item This is OLS through the origin: regress $Y_{\text{CA},t} - Y_{\text{NV},t}$
        on $Y_{\text{UT},t} - Y_{\text{NV},t}$, one observation per pre-period year.
  \item SSR is a parabola in $w$: if the solution falls outside $[0,1]$, take the
        nearer endpoint.
\end{itemize}
\end{frame}

\begin{frame}{Two Donors on Prop 99}
$w = 123{,}200 / 214{,}460 = 0.574$ \srcp{figs/w11\_scalar.R}.
\vskip-2pt
\begin{center}
\includegraphics[width=0.66\linewidth]{figs/two_donor.pdf}
\end{center}
\vskip-9pt
\begin{itemize}\itemsep1pt\small
  \item SSR: $w=0$ (Nevada only) 70{,}976; $w=1$ (Utah only) 39{,}037;
        $w=0.5$ 1{,}392; $w=0.574$ 202. Neither state alone resembles California;
        the mixture does.
  \item Pre-period fit in packs: root mean squared prediction error,
        $\text{RMSPE} = \sqrt{\text{SSR}/19} = \sqrt{202/19} = 3.26$, the typical
        size of a pre-period gap.
  \item Effect: average gap 1989--2000 $= -15.8$ packs, against $-27.3$ from the
        donor average.
\end{itemize}
\end{frame}

\begin{frame}{Many Donors\core}
Same objective, one weight per donor:
\begin{align*}
  \min_{w_2,\dots,w_{J+1}}\ \sum_{t \leq T_0}\Big(Y_{1t} - \sum_{j=2}^{J+1} w_j Y_{jt}\Big)^2
  \qquad\text{subject to}\quad w_j \geq 0,\ \ \sum_{j=2}^{J+1} w_j = 1 .
\end{align*}
\vskip3pt
\begin{itemize}\itemsep3pt
  \item The same two rules as with Utah and Nevada, now on every weight: no
        weight is negative, and the weights add to one. With two donors the
        weights $w$ and $1-w$ already add to one, so the rules left only
        $0 \leq w \leq 1$.
  \item With 38 donors there is no formula like the one for $w$; software
        finds the weights (a quadratic program).
  \item On Prop 99, all 19 pre-years, 38 donors: average gap 1989--2000 $= -19.5$,
        pre-period RMSPE 1.66 \srcp{figs/w11\_prop99.R}.
  \item Without the constraints, 38 weights could fit 19 years exactly; more on this
        below.
\end{itemize}
\end{frame}

\begin{frame}{Why the Constraints Matter\core}
The restrictions $w_j\geq 0$ and $\sum_j w_j = 1$ look like technicalities. They
are the heart of the method.
\vskip4pt
\begin{itemize}\itemsep3pt
  \item \textbf{No extrapolation}: the synthetic unit is a \emph{convex
        combination} of real units, so it lies inside the donor pool's support.
        Unconstrained regression would happily assign weights of $+3$ and $-2$ and
        predict outside anything ever observed.
  \item \textbf{Sparsity}: the constraints usually produce weights on a handful
        of donors and zero on the rest. You can print the weight table and let
        readers judge whether the comparison is sensible.
  \item \textbf{Transparency}: the counterfactual is a named, weighted set of real
        places, not a regression coefficient.
  \item If no convex combination matches the treated unit, if California is
        richer than every donor, \alert{the method tells you}, through poor
        pre-treatment fit. That is a feature.
\end{itemize}
\end{frame}

\begin{frame}{What the Constraints Prevent, on Prop 99}
Drop $w_j \geq 0$ and keep $\sum_j w_j = 1$: with 38 donors and 19 pre-years,
infinitely many weightings fit the pre-period exactly. Take the smallest one
(minimum norm). Both columns fit the 19 pre-period outcomes \srcp{figs/w11\_prop99.R}:
\vskip4pt
\begin{center}\small
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}lcc@{}}
\toprule
 & SC weights & regression weights \\
\midrule
nonzero donors & 6 & 38 \\
negative weights & 0 & 17 (as low as $-0.13$, Tennessee) \\
pre-period fit & RMSPE 1.66 & exact \\
average gap 1989--2000 & $-19.5$ & $-15.5$ \\
\bottomrule
\end{tabular}
\end{center}
\vskip4pt
\begin{itemize}\itemsep2pt
  \item An exact fit with $J > T_0$ is guaranteed and therefore uninformative: it
        fits the noise.
  \item Negative weights mean the counterfactual California is ``West Virginia
        plus Connecticut minus Tennessee minus Mississippi'': extrapolation no
        reader can inspect \srcp{Abadie 2021}.
\end{itemize}
\end{frame}

\begin{frame}{Matching Predictors as Well as Past Outcomes\core}
\srct{Abadie, Diamond, and Hainmueller (2010)} match \emph{predictors}
$k = 1,\dots,K$. Each $X_{jk}$ is \emph{one number} per state that the researcher
defines: mean price 1980--88, sales in 1975, \dots\par\vskip3pt
\textbf{Inner step.} Fix predictor weights $v_k \geq 0$. Choose $w$ to match the
predictors:
\begin{align*}
  w(v) = \argmin_{w_2,\dots,w_{J+1}}\ \textstyle\sum_{k=1}^{K} v_k\big(X_{1k} - \sum_{j=2}^{J+1} w_j X_{jk}\big)^2,
  \qquad w_j \geq 0,\ \ \textstyle\sum_j w_j = 1 .
\end{align*}
\textbf{Outer step.} Among all choices of $v$, keep the one whose $w(v)$ fits the
pre-period \emph{outcomes} best:
\begin{align*}
  \min_{v}\ \textstyle\sum_{t \leq T_0}\big(Y_{1t} - \sum_{j} w_j(v)\, Y_{jt}\big)^2 .
\end{align*}
\begin{itemize}\itemsep2pt
  \item $w$ minimizes the \emph{predictor} gaps; the outcome fit only chooses $v$.
  \item Two kinds of years: the outer step uses \emph{every} pre-year, 1970--88;
        predictors summarize windows the researcher picks (\alert{a choice}).
\end{itemize}
\end{frame}

\begin{frame}{The Two Steps on Two Donors\core}
Utah and Nevada again; two covariates \srcp{figs/w11\_scalar.R}: price, 1980--88
(CA 89.42, UT 89.43, NV 93.24); beer sales, 1984--88 (CA 24.28, UT 13.34, NV 37.00).
\vskip2pt
\begin{center}\small
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}ccc@{}}
\toprule
$v_{\text{price}}$ ($v_{\text{beer}} = 1 - v_{\text{price}}$) & inner $w(v)$ & outcome SSR \\
\midrule
0 \ (beer only) & 0.538 & 494 \\
0.5 & 0.549 & 337 \\
\textbf{0.768} & \textbf{0.574} & \textbf{202} \\

\bottomrule
\end{tabular}\\[1pt]
{\footnotesize $w$ = Utah's weight; Nevada gets $1 - w$.}
\end{center}
\vskip0pt
\begin{itemize}\itemsep2pt
  \item The loop: for $v_{\text{price}} = 0, .001, \dots, 1$, get $w(v)$, then the

  \item Price only: Utah has California's price, so $w = 1$. Beer only: $w = 0.538$.

\end{itemize}
{\footnotesize $v$ depends on units (price varies little, so it needs a big $v$):
a tuning knob, not importance.}
\end{frame}

the outer step: Utah .34, Nevada .24, Montana .21, Colorado .15, Connecticut .06,
applied to every year \srcp{figs/w11\_prop99.R}.
\vskip-2pt
\begin{center}
\includegraphics[width=0.72\linewidth]{figs/prop99_paths.pdf}
\end{center}
\vskip-9pt
\begin{itemize}\itemsep1pt
  \item Tracks California to 1988 (RMSPE 1.78), then separates. Average gap

  \item Outcomes-only weights (Many Donors slide) give $-19.5$; DiD gave $-27.3$.

\begin{frame}{Searching for $v$ When There Are Many Predictors}
With two predictors, $v$ had one free number and we tried 1,001 values. With
ADH's seven, $v$ has six free numbers (they add to 1).
\vskip2pt
\begin{itemize}\itemsep3pt
  \item A grid of 1,000 values each would be $10^{18}$ tries. Software
        \alert{searches}: move $v$ in the direction that lowers SSR until it
        stops falling.
  \item The search can stop at a \emph{local} minimum; different software
        reports different $v$ and $w$ for the same data \srcp{Klößner et al.\
        2018}. Report $w$ and the fit; do not read $v_k$ as importance.
  \item A common way out: drop the covariates and fit $w$ to the pre-period
        outcomes, as on the Many Donors slide. No $v$, nothing to search.
        Prop 99: SSR 52 (ADH's 60), gap $-19.5$.
  \item Covariates guard against fitting noise in the outcome path; past
        outcomes guard against unmeasured traits. Report both when they differ.
\end{itemize}
\end{frame}

\begin{frame}{What Have We Assumed?\core}
Everything so far used 1970--88 only. After 1988 we keep the same weights:
\begin{align*}
  \hat Y_{\text{CA},1995}(0) &= .342\,Y_{\text{UT},1995} + .238\,Y_{\text{NV},1995}
  + .209\,Y_{\text{MT},1995} \\ &\quad + .149\,Y_{\text{CO},1995} + .062\,Y_{\text{CT},1995} .
\end{align*}
\textbf{Assumption}: a mix that tracked California for 19 years would have kept
tracking it.
\vskip4pt
Why would a mix track California before 1989? Two possible reasons.
\begin{enumerate}\itemsep3pt
  \item \textbf{It shares what drives California's sales}: income, prices,
        culture, how health-conscious people are. Then it keeps tracking after
        1988. \alert{This is what we hope.}
  \item \textbf{Luck}: with 38 donors, some mix can cancel California's
        year-to-year bumps. Those bumps are new after 1988, so the tracking stops.
\end{enumerate}
\end{frame}

\begin{frame}{Synthetic Control Is a 2$\times$2 With a Built Control\core}
DiD and SC both compare California's change with a control's change, before
(1970--88) and after (1989--2000). Mean packs per capita; differences use
unrounded means \srcp{figs/w11\_2x2.R}.
\vskip1pt
\begin{center}\small
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}lccc@{}}
\toprule
 & before & after & after $-$ before \\
\midrule
California & 116.2 & 60.4 & \\
Control 1: average of 38 donors (\emph{given}) & 130.6 & 102.1 & \\
\quad gap, California $-$ control 1 & $-14.4$ & $-41.7$ & $\mathbf{-27.3}$ \ (DiD) \\
\midrule
Control 2: synthetic California (\emph{built}) & 116.0 & 79.2 & \\
\quad gap, California $-$ control 2 & $0.2$ & $-18.8$ & $\mathbf{-19.1}$ \ (SC) \\
\bottomrule
\end{tabular}
\end{center}
\vskip-2pt
\begin{itemize}\itemsep2pt
  \item Same 2$\times$2, different control: DiD takes it as given; SC builds it
        so that the before gap is about 0.
  \item \alert{Parallel trends, for both}: without the law, the gap stays at its before level.
  \item SC's usual estimate, the after gap alone ($-18.8$), takes the before gap as 0.
\end{itemize}
\end{frame}

\vskip-2pt
\begin{center}
\includegraphics[width=0.82\linewidth]{figs/prop99_2x2.pdf}
\end{center}

  \item Right: the built control's gap is near 0 before 1989; the assumption is

\begin{frame}{When Parallel Trends Fails After Weighting\core}
Weighting does not remove the assumption: without the law, California and
synthetic California would have moved together after 1988. It fails when:
\begin{enumerate}\itemsep2pt
  \item \textbf{The fit was luck}: with many donors and a short pre-period, the
        weights match noise, not what drives sales.
  \item \textbf{No mix reaches California}: it is extreme; the fit fails before 1989.
  \item \textbf{A donor has its own shock after 1988}: e.g.\ its own cigarette tax.
  \item \textbf{The law reaches the donors}: Californians buy cigarettes in
        Nevada (weight .24), so Nevada's sales rise because of Prop 99.
        Synthetic California is then too high: the effect is overstated.
  \item \textbf{California reacts early}: once the law is known, before it starts.
  \item \textbf{Something else changes in California at the same time}: SC
        estimates all of it. Prop 99 was itself a tax rise \emph{and} an
        anti-smoking program; SC cannot split them.
\end{enumerate}
\end{frame}

\begin{frame}{What to Check, What to Do\core}
\begin{itemize}\itemsep3pt
  \item \textbf{Choose the donors first}: drop states with their own tobacco
        policies (ADH dropped 11 states and DC) and neighbours the law may reach.
        Then accept the fit; do not shop for the pool that fits best.
  \item \textbf{Pre-period fit}: plot both paths for 1970--88; report the RMSPE
        in packs.
  \item \textbf{In-time placebo}: pretend the law began in 1981, fit the weights
        on 1970--80, and check that the 1981--88 gap is near 0.
  \item \textbf{Leave-one-out}: drop each donor with positive weight, refit; the
        estimate should barely move.
  \item \textbf{Anticipation}: a law can matter once it is known. Prop 99 passed
        in November 1988 and the tax rose in January 1989, so smokers could stock
        up first. If the gap opens early, start the treatment at the vote.
\end{itemize}
\end{frame}


\begin{frame}{Inference: the Placebo Test\core}
One treated state, so no usual standard error. Instead, run the method on every
state \srcp{Abadie, Diamond, and Hainmueller 2010}:
\begin{enumerate}\itemsep2pt
  \item Treat each of the 39 states in turn as ``treated''; build its synthetic
        control from the other 38.
  \item Compute its ratio $\text{RMSPE}^{\text{after}} / \text{RMSPE}^{\text{before}}$.
        Dividing by the pre-period fit keeps a state the method could not fit
        from looking like a big effect.
  \item Rank California among the 39: $p = \text{rank}/39$.
\end{enumerate}
\vskip3pt
\textbf{Can it be reported as a $p$-value?} Yes: it is a permutation test, and
papers report it as the $p$-value.
\begin{itemize}\itemsep2pt
  \item Meaning: if the law did nothing and California were one of 39
        interchangeable states, a ratio this large would come up 1 time in 39.
        Exact only if the treated state had been picked at random.
  \item Always state the floor: with 39 states the smallest possible $p$ is
        $1/39 = 0.026$.
\end{itemize}
\end{frame}

\begin{frame}{A Standard Error From the Placebos}
Second option \srcp{Arkhangelsky et al.\ 2021, \texttt{synthdid}}: estimate the
``effect'' for placebo states, and use their spread as the standard error.
\begin{align*}
  \text{SE} = \text{sd of the placebo estimates}, \qquad
  \text{95\% CI} = \hat\tau \pm 1.96\,\text{SE} .
\end{align*}
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lccc@{}}
\toprule
Prop 99 & estimate & placebo SE & 95\% CI \\
\midrule
SC & $-19.6$ & 11.1 & $[-41.4,\ 2.1]$ \\
SDiD & $-15.6$ & 8.9 & $[-33.1,\ 1.9]$ \\
\bottomrule
\end{tabular}
\end{center}
{\footnotesize\srcp{figs/w11\_prop99.R}}
\begin{itemize}\itemsep2pt
  \item The SC interval includes 0, while the rank test gives $p = 0.026$. The
        SE uses the raw placebo effects, so states the method fits badly inflate
        it; the ratio discounts them.
  \item Report the rank $p$-value; add the placebo SE if you want an interval,
        and say which one you used.
\end{itemize}
\end{frame}

\begin{frame}{In the Literature: Bartos and Kubrin (2018)}
\textbf{The question.} California's Proposition 47 (November 2014) reclassified
some drug and property felonies as misdemeanors, shrinking prison and jail
populations. Did crime rise? Opponents pointed to a 2015 uptick.
\vskip3pt
\textbf{The data.} FBI Uniform Crime Reports, state rates per capita,
1970--2015: seven offenses, each with its own synthetic California. 44 pre-period
years; one post-period year (2015).
\vskip3pt
\textbf{The design.} Donor pool: the other 49 states, since none adopted a
comparable reform. Weights from pre-period crime trends. Effect: the 2015 gap.
\vskip3pt
\textbf{Decision rule.} A gap smaller than the pre-period RMSPE is treated as
no effect; a placebo rank in the top 5 of the 50 states (share 0.10) counts as evidence.
\end{frame}

\section[Extensions]{Extensions}

\begin{frame}\sectionpage\end{frame}

\begin{frame}{Penalized Synthetic Control \srcp{Abadie and L'Hour 2021}}
Penalize the distance to every donor used (sums over predictors $k = 1,\dots,K$
and donors $j = 2,\dots,J+1$; $w_j \geq 0$, $\sum_j w_j = 1$):
\begin{align*}
  w^{\text{pen}} = \argmin_{w_2,\dots,w_{J+1}}\
  \underbrace{\textstyle\sum_{k}\big(X_{1k} - \sum_{j} w_j X_{jk}\big)^2}_{\text{the mix vs.\ California}}
  \; + \; \lambda \textstyle\sum_{j} w_j
  \underbrace{\textstyle\sum_{k}\big(X_{1k} - X_{jk}\big)^2}_{\text{donor $j$ vs.\ California}} .
\end{align*}
Hypothetical, $K = 1$: $X_1 = 5$; donors A $= 4$, B $= 6$, C $= 0$, D $= 10$
\srcp{figs/w11\_scalar.R}.
\begin{center}\small
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}lccc@{}}
\toprule
mix & $\sum_j w_j X_{j}$ & mix vs.\ California & penalty $\sum_j w_j (X_{1} - X_{j})^2$ \\
\midrule
$\tfrac12$A $+ \tfrac12$B & 5 & 0 & $\tfrac12\cdot 1 + \tfrac12\cdot 1 = 1$ \\
$\tfrac12$C $+ \tfrac12$D & 5 & 0 & $\tfrac12\cdot 25 + \tfrac12\cdot 25 = 25$ \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}\itemsep2pt
  \item Both fit perfectly; plain SC cannot choose. Any $\lambda > 0$ picks the nearby A, B.
  \item $\lambda \to 0$: SC; $\lambda \to \infty$: nearest-neighbour matching (Week 6).
  \item \texttt{pensynth::cv\_pensynth} picks $\lambda$ by pre-period fit. Prop 99:
        $\lambda \approx 0$, the same weights as SC (Lab 11).
\end{itemize}
\end{frame}

\begin{frame}{Augmented Synthetic Control \srcp{Ben-Michael, Feller, and Rothstein 2021}}
\textbf{Idea}: if synthetic California misses California before 1989, part of
the later gap is that mismatch, not the law. Predict it and subtract it. $T$: one
post-treatment year (1989, \dots, 2000); repeat for each and average.
\begin{enumerate}\itemsep2pt
  \item Build SC as usual: weights $w_j$.
  \item Pool the 38 donors and regress year-$T$ sales on each state's own 19
        pre-period sales, $Y_{jT} \approx \eta_0 + \sum_{t \leq 1988}\eta_t Y_{jt}$
        (ridge; no unit or year effects).
  \item Predict each state from its own past: $\hat m_{jT}$ (donors), $\hat m_{1T}$ (California).
  \item Subtract the gap their pasts alone predict:
\end{enumerate}
\vskip-8pt
\begin{align*}
  \text{effect}_T = \underbrace{\Big(Y_{1T} - \textstyle\sum_j w_j Y_{jT}\Big)}_{\text{SC gap}}
   - \underbrace{\Big(\hat m_{1T} - \textstyle\sum_j w_j \hat m_{jT}\Big)}_{\text{gap the pasts predict}} .
\end{align*}
\vskip-4pt
\begin{itemize}\itemsep1pt
  \item Perfect pre-period fit: same pasts, same predictions, nothing subtracted.
  \item Prop 99 (\texttt{augsynth}): SC gap $-19.5$, predicted gap $-3.6$,
        augmented $-16.0$ \srcp{figs/w11\_prop99.R}.
\end{itemize}
\end{frame}

\begin{frame}{Synthetic DiD: Match the Trend, Not the Level}
\textbf{SC} needs California's \emph{level} in every pre-year; \textbf{DiD} needs
only its \emph{trend}. \textbf{Synthetic DiD} \srcp{Arkhangelsky et al.\ 2021}
builds the control like SC but asks only for the trend:
\begin{align*}
  \min_{\omega,\ c}\ \textstyle\sum_{t\leq 1988}\Big(Y_{1t} - c - \sum_j \omega_j Y_{jt}\Big)^2,
  \qquad \omega_j \geq 0,\ \textstyle\sum_j\omega_j = 1 .
\end{align*}
The free constant $c$ lets the control sit above or below California: only the
\emph{shape} of the path has to match. What this buys:
\begin{itemize}\itemsep1pt
  \item \textbf{Extreme units}: if California had the lowest sales, no average of
        donors could reach its level, but a mix can still share its trend.
  \item \textbf{More mixes qualify}, so the weight can spread over many states
        (a small penalty encourages this).
  \item \textbf{The cost}: the level gap is assumed constant, so we are back to
        DiD's parallel trends, now for a weighted control.
\end{itemize}
\end{frame}

\begin{frame}{Synthetic DiD on Prop 99}
Two sets of weights, then the 2$\times$2 \srcp{figs/w11\_prop99.R}:
\begin{itemize}\itemsep2pt
  \item \textbf{Donor weights} $\omega_j$, fitted with the free constant $c$:
        Nevada .12, New Hampshire .11, Connecticut .08, \dots
  \item \textbf{Pre-year weights} $\lambda_t$: the ``before'' period is the
        pre-years that best predict the donors' post-period mean: 1986--88
        (.37, .21, .43). The level gap is measured on years like the after period.
\end{itemize}
\vskip3pt
\begin{center}\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}lccc@{}}
\toprule
 & before (weighted) & after & after $-$ before \\
\midrule
California & 95.1 & 60.4 & \\
SDiD control & 120.5 & 101.3 & \\
gap & $-25.4$ & $-41.0$ & $\mathbf{-15.6}$ \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}\itemsep2pt
  \item The control sits 25 packs above California before 1989. SC would not
        allow that; SDiD does, because only the trend has to match.
  \item DiD $-27.3$, SC $-19.6$, SDiD $-15.6$ (\texttt{synthdid}, as in their Table 1).
\end{itemize}
\end{frame}

\begin{frame}{Choosing Among Them}
\begin{center}\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lll}
\toprule
Situation & Method & \texttt{R} \\
\midrule
One treated unit, good pre-fit & classic synthetic control & \texttt{Synth}, \texttt{tidysynth} \\
One treated unit, poor pre-fit & augmented / ridge SC & \texttt{augsynth} \\
Many units, want DID robustness & synthetic DID & \texttt{synthdid} \\
Many units, staggered, covariates & Callaway--Sant'Anna (Wk 10) & \texttt{did} \\
\bottomrule
\end{tabular}
\end{center}
\vskip5pt
\begin{itemize}\itemsep2pt
  \item These are not rivals so much as points on a spectrum between ``trust the
        weights'' and ``trust the outcome model.'' The augmented versions
        deliberately use both, the same insurance logic as AIPW.
  \item Report the classic estimator alongside whichever you prefer, so readers
        can see how much the extrapolation is doing.
\end{itemize}
\end{frame}
